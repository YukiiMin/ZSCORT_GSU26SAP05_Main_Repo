CLASS zcl026_scort_release_service DEFINITION
  PUBLIC
  FINAL
  CREATE PUBLIC .

  PUBLIC SECTION.
    CLASS-METHODS process_release
      IMPORTING
        iv_trkorr       TYPE trkorr
        iv_dialog       TYPE abap_bool DEFAULT abap_false
      EXPORTING
        ev_success      TYPE abap_bool
        ev_request_type TYPE string
        ev_status       TYPE trstatus
        ev_message      TYPE string.
ENDCLASS.
CLASS zcl026_scort_release_service IMPLEMENTATION.

  METHOD process_release.
    DATA lv_fm_subrc TYPE sysubrc.
    DATA lv_try      TYPE i.

    ev_success = abap_false.

    SELECT SINGLE trkorr, strkorr, trstatus FROM e070
      WHERE trkorr = @iv_trkorr
      INTO @DATA(ls_e070).

    IF sy-subrc <> 0.
      ev_status  = 'UNKNOWN'.
      ev_message = |Transport Request/Task { iv_trkorr } không tồn tại.|.
      RETURN.
    ENDIF.

    ev_request_type = COND #( WHEN ls_e070-strkorr IS NOT INITIAL THEN 'TASK' ELSE 'REQUEST' ).
    ev_status       = ls_e070-trstatus.

    " Đã Released
    IF ls_e070-trstatus = 'R' OR ls_e070-trstatus = 'N'.
      ev_success = abap_true.
      ev_message = |{ COND string( WHEN ev_request_type = 'TASK' THEN 'Task' ELSE 'TR' ) } { iv_trkorr } đã Released (status={ ls_e070-trstatus }).|.
      RETURN.
    ENDIF.

    " O = Release started / đang export — không gọi CTS lại
    IF ls_e070-trstatus = 'O'.
      DO 5 TIMES.
        lv_try = lv_try + 1.
        WAIT UP TO 1 SECONDS.
        SELECT SINGLE trstatus FROM e070 WHERE trkorr = @iv_trkorr INTO @ev_status.
        IF ev_status = 'R' OR ev_status = 'N'.
          ev_success = abap_true.
          ev_message = |Export xong: { iv_trkorr } → status { ev_status }.|.
          RETURN.
        ENDIF.
      ENDDO.
      ev_success = abap_true.
      ev_message = |{ iv_trkorr } đang export (status O). Chờ SE09/STMS xong → R, rồi Search lại. Compare/Apply cần R.|.
      RETURN.
    ENDIF.

    CALL FUNCTION 'TR_RELEASE_REQUEST'
      EXPORTING
        iv_trkorr             = iv_trkorr
        iv_dialog             = iv_dialog
        iv_display_export_log = abap_false
        iv_success_message    = abap_false
      EXCEPTIONS
        cts_initialization_failure = 1
        enqueue_failed             = 2
        no_authorization           = 3
        invalid_request            = 4
        request_already_released   = 5
        repeat_unable              = 6
        object_check_error         = 7
        object_conversion_error    = 8
        model_check_error          = 9
        released_with_warning      = 10
        released_with_error        = 11
        information_to_display     = 12
        cancelled                  = 13
        OTHERS                     = 14.

    lv_fm_subrc = sy-subrc.

    IF lv_fm_subrc = 0 OR lv_fm_subrc = 10 OR lv_fm_subrc = 12.
      " Chờ export (Request thường D→O→R)
      CLEAR lv_try.
      DO 8 TIMES.
        lv_try = lv_try + 1.
        SELECT SINGLE trstatus FROM e070 WHERE trkorr = @iv_trkorr INTO @ev_status.
        IF ev_status = 'R' OR ev_status = 'N'.
          EXIT.
        ENDIF.
        WAIT UP TO 1 SECONDS.
      ENDDO.
      SELECT SINGLE trstatus FROM e070 WHERE trkorr = @iv_trkorr INTO @ev_status.

      IF ev_status = 'R' OR ev_status = 'N'.
        ev_success = abap_true.
        IF ev_request_type = 'TASK'.
          ev_message = |Release Task { iv_trkorr } OK. Object merge về TR { ls_e070-strkorr }.|.
        ELSE.
          ev_message = |Release TR { iv_trkorr } OK (status { ev_status }).|.
        ENDIF.
      ELSEIF ev_status = 'O'.
        " CTS đã nhận release — export chưa xong (bình thường với TR cha)
        ev_success = abap_true.
        ev_message = |Release { iv_trkorr } đã start (status O). Chờ export SE09 → R rồi Search. Compare/Apply cần R.|.
      ELSE.
        ev_success = abap_false.
        ev_message = |Release { iv_trkorr } chưa xong (status={ ev_status }, FM={ lv_fm_subrc }). Xem SE09.|.
      ENDIF.
      RETURN.
    ENDIF.

    ev_success = abap_false.
    CASE lv_fm_subrc.
      WHEN 1.  ev_message = |Release { iv_trkorr }: CTS init failed|.
      WHEN 2.  ev_message = |Release { iv_trkorr }: locked — SE03 unlock / thử lại|.
      WHEN 3.  ev_message = |Release { iv_trkorr }: no authorization|.
      WHEN 4.  ev_message = |Release { iv_trkorr }: invalid request|.
      WHEN 5.  ev_message = |Release { iv_trkorr }: already released|.
      WHEN 6.  ev_message = |Release { iv_trkorr }: cannot repeat yet|.
      WHEN 7.  ev_message = |Release { iv_trkorr }: object check — SE09 Release xem log (không chỉ Activate; có thể $TMP/ATC/syntax)|.
      WHEN 8.  ev_message = |Release { iv_trkorr }: object conversion error|.
      WHEN 9.  ev_message = |Release { iv_trkorr }: model check error|.
      WHEN 11. ev_message = |Release { iv_trkorr }: released with error — SE09 export log|.
      WHEN 13. ev_message = |Release { iv_trkorr }: cancelled|.
      WHEN OTHERS.
        ev_message = |Release { iv_trkorr } failed (subrc { lv_fm_subrc })|.
    ENDCASE.
  ENDMETHOD.

ENDCLASS.
